# Vanguard Preloader Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a zero-allocation, ultra-fast priority asset preloader for Roblox with "Plug & Play" developer QoL features.

**Architecture:** A central facade (`init.luau`) delegating to a `Queue` (pre-allocated flat arrays with read/write cursors) and a `State` dictionary for O(1) deduplication. A single background `Worker` handles batching to `ContentProvider`, with auto-formatting and crash prevention at the entry point.

**Tech Stack:** Luau, Roblox Studio, Aftman, Rojo.

---

### Scope Note
To strictly adhere to the rule of providing *complete, working code without placeholders*, this plan covers **Vanguard Core Phase 1**: Infrastructure, 0-Allocation Queues, State Management, and the Public API with Quality of Life features (Anti-Crash, Auto-Format, OnReady, Timeout). Advanced modules (Spatial Priority, Throttling) should be implemented in subsequent plans once the Core is fully tested.

```luau
-- src/Core/Queue.luau
--!strict
--!native
--!optimize 2

local Queue = {}
local Capacities = {1000, 2000, 5000, 5000}
local Queues = {
    [1] = table.create(Capacities[1]), -- Critical
    [2] = table.create(Capacities[2]), -- High
    [3] = table.create(Capacities[3]), -- Normal
    [4] = table.create(Capacities[4])  -- Background
}

local WriteHeads = {1, 1, 1, 1}
local ReadHeads = {1, 1, 1, 1}
local TotalPushed = {0, 0, 0, 0}
local TotalLoaded = {0, 0, 0, 0}

function Queue.Push(priority: number, asset: any)
    local head = WriteHeads[priority]
    Queues[priority][head] = asset
    -- Modulo wrap-around for true Ring Buffer (1-based index)
    WriteHeads[priority] = (head % Capacities[priority]) + 1
    TotalPushed[priority] += 1
end

function Queue.Pop(priority: number): any
    local readHead = ReadHeads[priority]
    local writeHead = WriteHeads[priority]
    
    if readHead ~= writeHead then
        local asset = Queues[priority][readHead]
        Queues[priority][readHead] = nil -- Free reference for GC (Memory Leak prevention)
        ReadHeads[priority] = (readHead % Capacities[priority]) + 1
        return asset
    end
    return nil
end

function Queue.IsEmpty(priority: number): boolean
    return ReadHeads[priority] == WriteHeads[priority]
end

function Queue.AddLoaded(priority: number, count: number)
    TotalLoaded[priority] += count
end

function Queue.GetProgress(priority: number): (number, number)
    return TotalLoaded[priority], TotalPushed[priority]
end

function Queue.ResetTracking(priority: number)
    TotalPushed[priority] = 0
    TotalLoaded[priority] = 0
end

function Queue.Clear(priority: number)
    ReadHeads[priority] = WriteHeads[priority]
    table.clear(Queues[priority])
end

return Queue
```

```luau
-- src/Core/Worker.luau
--!strict
local ContentProvider = game:GetService("ContentProvider")
local Queue = require(script.Parent.Queue)
local State = require(script.Parent.State)

local Worker = {}
local isRunning = false
local isPaused = false
local VanguardDebug = false

local OnReadyCallbacks: { [number]: { () -> () } } = { {}, {}, {}, {} }
local OnProgressCallbacks: { [number]: { (number, number) -> () } } = { {}, {}, {}, {} }
local OnFailedCallbacks: { [number]: { () -> () } } = { {}, {}, {}, {} }

local BATCH_SIZE = 50
local NextBatchSize = BATCH_SIZE
local BatchArray = table.create(BATCH_SIZE)

local function ProcessQueue()
    while true do
        if isPaused then
            task.wait(0.1)
            continue
        end
        
        if NextBatchSize ~= BATCH_SIZE then
            BATCH_SIZE = NextBatchSize
            BatchArray = table.create(BATCH_SIZE)
        end
        
        local processedAny = false
        
        for priority = 1, 4 do
            if not Queue.IsEmpty(priority) then
                processedAny = true
                local batchCount = 0
                
                while batchCount < BATCH_SIZE and not Queue.IsEmpty(priority) do
                    local asset = Queue.Pop(priority)
                    -- Lazy deletion: skip if state is no longer 1 (e.g. cancelled or loaded)
                    if asset and State.Get(asset) == 1 then
                        batchCount += 1
                        BatchArray[batchCount] = asset
                    end
                end
                
                if batchCount == 0 then break end -- Skip and restart prioritization
                
                -- Slice array to exact batch size (ContentProvider requires exact array)
                local exactBatch = table.move(BatchArray, 1, batchCount, 1, table.create(batchCount))
                local currentThread = coroutine.running()
                
                local success = false
                local retries = 0
                
                -- Auto-Retry Loop (1 retry max)
                while retries <= 1 and not success do
                    local loadThread = task.spawn(function()
                        local pcallSuccess = pcall(function()
                            ContentProvider:PreloadAsync(exactBatch)
                        end)
                        task.spawn(currentThread, pcallSuccess) 
                    end)
                    
                    local timeoutThread = task.delay(10, function()
                        if coroutine.status(loadThread) ~= "dead" then
                            task.cancel(loadThread)
                            task.spawn(currentThread, false)
                        end
                    end)
                    
                    success = coroutine.yield()
                    if success then
                        task.cancel(timeoutThread)
                    else
                        retries += 1
                        if VanguardDebug then warn("Vanguard: Timeout on priority " .. priority .. " (Retry " .. retries .. ")") end
                    end
                end
                
                if success then
                    Queue.AddLoaded(priority, batchCount)
                    for i = 1, batchCount do State.Set(exactBatch[i], 2) end
                    
                    local loaded, total = Queue.GetProgress(priority)
                    for _, cb in ipairs(OnProgressCallbacks[priority]) do task.spawn(cb, loaded, total) end
                else
                    if VanguardDebug then warn("Vanguard: Abandoning batch on priority " .. priority, exactBatch) end
                    for i = 1, batchCount do State.Set(exactBatch[i], 3) end 
                    for _, cb in ipairs(OnFailedCallbacks[priority]) do task.spawn(cb) end
                end
                
                -- Trigger OnReady if queue is now empty
                if Queue.IsEmpty(priority) then
                    for _, callback in ipairs(OnReadyCallbacks[priority]) do task.spawn(callback) end
                    table.clear(OnReadyCallbacks[priority])
                    Queue.ResetTracking(priority)
                end
                
                break -- Go back to priority 1 to ensure criticals are respected
            end
        end
        
        if not processedAny then
            task.wait(0.1) -- Idle sleep
        end
    end
end

function Worker.Start()
    if isRunning then return end
    isRunning = true
    task.spawn(ProcessQueue)
end

function Worker.Pause() isPaused = true end
function Worker.Resume() isPaused = false end
function Worker.SetDebug(value: boolean) VanguardDebug = value end

function Worker.SetBatchSize(size: number)
    NextBatchSize = size
end

function Worker.RegisterOnReady(priority: number, callback: () -> ()) table.insert(OnReadyCallbacks[priority], callback) end
function Worker.RegisterOnProgress(priority: number, callback: (number, number) -> ()) table.insert(OnProgressCallbacks[priority], callback) end
function Worker.RegisterOnFailed(priority: number, callback: () -> ()) table.insert(OnFailedCallbacks[priority], callback) end

return Worker
```

```luau
-- src/init.luau
--!strict
local Queue = require(script.Core.Queue)
local State = require(script.Core.State)
local Worker = require(script.Core.Worker)

local Vanguard = {}

-- Auto-start worker
Worker.Start()

local function FormatAsset(asset: any): any
    if type(asset) == "number" then
        return "rbxassetid://" .. tostring(asset)
    elseif type(asset) == "string" then
        if string.match(asset, "^%d+$") then
            return "rbxassetid://" .. asset
        end
        return asset
    elseif typeof(asset) == "Instance" then
        -- Direct Instance support for ContentProvider (avoids nil block)
        return asset
    end
    return nil
end

function Vanguard.Load(asset: any, priority: number?)
    local prio = priority or 3
    local formattedAsset = FormatAsset(asset)
    
    -- Anti-Crash & Silent Skip
    if not formattedAsset then return end
    
    local currentState = State.Get(formattedAsset)
    if currentState > 1 then return end -- Already Loaded(2) or Failed(3)
    
    -- Priority Upgrading & Initial Push (Lazy Deletion handles duplicates)
    State.Set(formattedAsset, 1) -- Set Pending
    Queue.Push(prio, formattedAsset)
end

function Vanguard.Cancel(asset: any)
    local formatted = FormatAsset(asset)
    if formatted and State.Get(formatted) == 1 then
        State.Set(formatted, 0)
    end
end

function Vanguard.OnReady(priority: number, callback: () -> ())
    if Queue.IsEmpty(priority) then
        task.spawn(callback)
    else
        Worker.RegisterOnReady(priority, callback)
    end
end

function Vanguard.OnProgress(priority: number, callback: (number, number) -> ())
    Worker.RegisterOnProgress(priority, callback)
end

function Vanguard.OnFailed(priority: number, callback: () -> ())
    Worker.RegisterOnFailed(priority, callback)
end

function Vanguard.Wait(priority: number)
    if Queue.IsEmpty(priority) then return end
    local thread = coroutine.running()
    Vanguard.OnReady(priority, function() task.spawn(thread) end)
    coroutine.yield()
end

function Vanguard.GetStatus(asset: any): number
    local formatted = FormatAsset(asset)
    return formatted and State.Get(formatted) or 0
end

function Vanguard.ClearQueue(priority: number)
    Queue.Clear(priority)
end

function Vanguard.SetBatchSize(size: number)
    Worker.SetBatchSize(size)
end

function Vanguard.LoadList(list: {any}, priority: number?)
    local prio = priority or 3
    for _, asset in ipairs(list) do
        Vanguard.Load(asset, prio)
    end
end

function Vanguard.WaitForAsset(asset: any): boolean
    local formatted = FormatAsset(asset)
    if not formatted then return false end
    
    local status = State.Get(formatted)
    if status == 2 then return true end
    if status == 3 then return false end
    
    if status == 0 then Vanguard.Load(formatted, 1) end
    
    while true do
        local s = State.Get(formatted)
        if s == 2 then return true end
        if s == 3 then return false end
        task.wait()
    end
end

function Vanguard.Pause() Worker.Pause() end
function Vanguard.Resume() Worker.Resume() end
function Vanguard.SetDebug(value: boolean) Worker.SetDebug(value) end

return Vanguard
```
